		{default: Turndown},
	] = await Promise.all(modules);

		XML: {ignoreAttributes: false, attributeNamePrefix: '@_'},
		HTML: {bulletListMarker: "-", emDelimiter: "*"},
	};
	const eidos = await app.vault
		.read(transDunamis()("cidoc_crm_v7.1.3.xml"))
		.then(d=>new XMLParser(tropos.XML).parse(d))

	const tropoKhora = ["crm_objects","crm_PoP"];
	tropoKhora.forEach(t=>eidos[t]=Object());
	
	const {
		classes: {["class"]: cidoc_class},
		properties: {property},
		...kytos
	} = eidos.cidoc_crm;
	eidos.cidoc_crm = kytos;
	
	for(const {["@_id"]: id, ...d} of [...cidoc_class, ...property]) {
		eidos[tropoKhora[0]][id] = d;
		
		[
			Object.keys(d).find(t=>t.startsWith("sub")),
			"domain",
			"range",
		].filter(Boolean).forEach(t=>d[t]&&=[d[t]].flat().map(p=>p["@_id"]));
		[
			"scopeNote",
			"examples",
			"inFirstOrderLogic",
			"quantification",
		].forEach(t=>d[t]&&=turndown.turndown(d[t]));
		
		if(Object.keys(d).includes("propertyOfProperty")){

				directName: khora.labelText,

				quantification: "(0,n:0,n)",
			};
		}
	}
	
	await Promise
		.resolve(JSON.stringify(eidos,null,"\t"))
		.then(t=>t.replaceAll("-   ", "- ").replaceAll("@_",""))
		.then(t=> app.vault.adapter.write("cidoc_crm_v7.1.3.json",t));
	return eidos;
};
