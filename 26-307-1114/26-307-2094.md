# あらまし
[[26-307-1114|結び付け方]]のための探り方。
受け取った `dunamis` から、 
1. 掛けられた[[26-307-1305|CRM Entity掛け]]の扱うCRM Classを求める
2. それぞれの Property を挙げ、上の場に用いられうるもののみを絞り込む
3. 辞書構造、その他参照しやすい形に仕立てる
# 仕掛け

### 仕組み方
```JS
const [
	{transDunamis, Mneme: {cascade, retrorse}},
	{getNoesei, Mneme: {tipos}},
	{toArray},
	relayTessera,
	pseudoTipos,
] = await syzygy('26-307-2094');
```
```JS
export default async function(noema,dunamis){
	const [
		khoroTipos,
		propTipos,
		typedTipos,
	] = cascade(this).slice(-3).map(d=>d.file.path);
	const isProperty = dunamis => [
		propTipos,
		typedTipos,
	].includes(tipos(dunamis)?.file?.path);
	
	if(this?.current)dunamis = this.current();
	const khora = await getNoesei(false)(dunamis)
		.then(d=>toArray(d.eidos??[]).map(k=>k?.[1]))
		.then(k=>k.filter(d=>tipos(d)?.file?.path===khoroTipos))
		.then(k=>new Set(k.map(transDunamis(true))));
	const noetoKhora = [
		d=>cascade(pseudoTipos(d)).filter(isProperty),
		async x =>{
			if(!Array.isArray(x))return; const [d,p] = x;
			const khoroNoema = new Set([...khora].map(k=>k[0]))
			const propKhora = await Promise
				.all(cascade(p).map(d=>relayTessera(d,false)))
				.then(k=>k.flat(Infinity))
				.then(k=>new Set(k.map(d=>d.file.path)))
			return [...khoroNoema].filter(d=>propKhora.has(d.file.path))
		},
	];
	
	for(const noesis of noetoKhora)
	for(const skia of [...khora]){
		const kytos = [skia].flat();
		const eidolon = await Promise
			.resolve(noesis(skia))
			.then(k=>k?.map?.(d=>[...kytos,d]))
		if(!eidolon.length)continue;
		else khora.add(...eidolon).delete(skia);
	};
	return [...khora]
}
```
