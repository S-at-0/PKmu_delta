# あらまし
[[26-307-1114|結び付け方]]のための探り方。
受け取った `dunamis` から、 
1. 掛けられた[[26-307-1305|結値掛け]]の扱うCRM Classを求める
2. それぞれの Property を挙げ、上の場に用いられうるもののみを絞り込む
3. 辞書構造、その他参照しやすい形に仕立てる

```JS
const [
	{transDunamis, Mneme: {cascade, retrorse}},
```

```JS
export default async function(noema,dunamis){
	if(this?.current)dunamis = this.current();
	
	const isKhorai =  hyle => cascade(tipos(hyle))
		.map(d=>d?.file?.path)
		.includes(cascade(this).pop().file.path);
	const kytos = await getNoesei(false)(dunamis)
		.then(Object.getPrototypeOf)
		.then(d=>(d.eidos??[]).map(k=>k?.[1]))
		.then(k=>k.map(d=>transDunamis()(d).path))
		.then(k=>k.filter(isKhorai));
	const khora = new Set(kytos);
	
	const kytoEidos = await Promise
		.all(kytos.map(d=>relayTessera(pseudoTipos(d),true)))
		.then(k=>k.map(x=>x.flat(Infinity)))
	const kytoTropos = kytoEidos
		.map(k=>k.flatMap(retrorse(cascade)))
		.map(k=>k.map(d=>d?.file?.path));
	const noetoKhora = [
		x=>getNoesei()(pseudoTipos(x))
			.then(Object.getPrototypeOf)
			.then(d=>(d?.eidos||[]).map(k=>k[1]).filter(isProperty))
			.then(k=>k.map(d=>transDunamis()(d).path)),
		x=>kytos.filter((_,i)=>kytoTropos[i].includes(x[1])),
	];
	console.log(noetoKhora);
	for(const noesis of noetoKhora)
	for(const skia of [...khora]){
		const eidolon = await noesis(skia)
		if(eidolon?.length)eidolon
			.map(d=>[skia].flat().concat(d))
			.forEach(k=>khora.add(k)) 
		khora.delete(skia);
		console.log("node",structuredClone(khora))
	};
	console.log("result",khora);
	return [...khora]
}
```
