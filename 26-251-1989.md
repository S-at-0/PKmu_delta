const noesis = async function(){
	const [
		{arche, Melete: {assignTo, getLinkText}},
		noesis,
		kytosTesserae,
		exoTally,
		kinetoKhora,
		ousioKinesis,
	] = await syzygy('26-251-1989');
	const noema = await noesis();
	
	const tessera = kytosTesserae();
	for(const eidos of Object.values(noema.crm_objects))eidos.tipos = 
		"className" in eidos? "class":
		"propertyOfProperty" in eidos? "typedProp":"property";
	Object.values(noema.crm_PoP).forEach(d=>d.tipos="PoP");
	Object.assign(noema.crm_objects,noema.crm_PoP);
	for(const eidos of Object.values(noema.crm_objects)){
		const morphe = {
			eidos: tessera[eidos.tipos],
			eidolon: "",
		}; eidos.dunamis = await exoTally(morphe);
		await assignTo(eidos.dunamis,arche.prop.title,eidos.fullName)
		const tropos = p =>p.aliases = [
			"className",
			"directName",
			"inverseName",
		].map(t=>eidos[t]).filter(Boolean);
		await app.fileManager.processFrontMatter(eidos.dunamis,tropos);
	}
	
	const message = "linkage constructuring: "
	const notice = new Notice(message,0);
	let counter = 0;
	
	for(const [kytos, eidos] of Object.entries(noema.crm_objects)){
		const hyloKytos = "あらまし"
		const hyle = [
			"examples",
			"inFirstOrderLogic",
			"quantification",
		].map(d=>d in eidos? `### ${d}\n${eidos[d]}`: null)
		
		eidos.scopeNote ??= "!" + await Promise
			.resolve(noema.crm_objects[eidos.domain].dunamis)
			.then(t=>getLinkText(t,hyloKytos))
			.catch(e=>e.message);
		hyle.unshift(hyloKytos,eidos.scopeNote);
		const tropos = t=>t.replace(hyloKytos,hyle.filter(Boolean).join("\n"));
		await app.vault.process(eidos.dunamis,tropos);
		
		const eidoKinesis = (...khora) => Promise
			.resolve(khora.flat(Infinity).filter(Boolean))
			.then(k=>k.map(t=>noema.crm_objects[t].dunamis))
			.then(([kytos,...k])=>[kytos,Promise.all(k.map(d=>getLinkText(d)))])
			.then(s=>Promise.all(s))
			.then(([kytos,k])=>assignTo(kytos,arche.prop.cascade,k));
		switch(eidos.tipos){
			case "class": 
				await eidoKinesis(kytos, eidos.subClassOf);
			break;
			case "property":
			case "typedProp":
				await eidoKinesis(kytos, eidos.subPropertyOf, eidos.range);
				await Promise.all(eidos.domain.map(t=>kinesis(t,kytos)))
				const hexis = {dunamis: eidos.dunamis};
				const [[delta,nu],[rho,kappa]] = eidos.quantification
					.match(/\((.*)\)/).replaceAll("n","0")
					.split(":").map(t=>t.split(",").map(Number))
					.map(([a,b])=>[a, b && b-a+1])
				Object.assign(hexis,{delta,nu,rho,kappa});
				hexis.tropoKhora = eidos.directName;
			break;
			case "PoP":
				const khoroNoema = await getNoesei(true)(tessera.PoP_domain)
				const tropoNoesis = await Promise
					.resolve(noema.crm_object[eidos.domain].dunamis)
					.then(d=>getLinkText(d))
					.then(t=>tropos(khoroNoema,t));
				const kinesis = await kinetoKhora(khoroNoema,eidos.dunamis)
					.then(k=> t=>t.replace(k,`${k}\n${tropoNoesis}`));
				await app.vault.process(eidos.dunamis,kinesis);
			break;
		}
		notice.setMessage(message + ++counter);
	};
	notice.setMessage(message + "done");
	notice.messageEl.style.color = "green";
};
if(typeof isModule === "undefined") noesis();
