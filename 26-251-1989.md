const noesis = async function(){
	const [
		{arche, Melete: {assignTo, getLinkText}},
		noesis,
		kytosTesserae,
		exoTally,
	] = await syzygy('26-251-1989');
	const noema = await noesis();
	
	for(const eidos of Object.values(noema.crm_objects)){
		const tessera = 
			"className" in eidos? "class":
			"propertyOfProperty" in eidos? "typedProp":"property";
		console.log(tessera)
		eidos.tipos = kytosTesserae()[tessera]
	}
	Object.values(noema.crm_PoP).forEach(d=>d.tipos=kytosTesserae().PoP)
	
	for(const eidos of Object.values({...noema.crm_objects,...noema.crm_PoP})){
		eidos.dunamis = await exoTally({eidos: eidos.tipos,eidolon:""})
		{
			await assignTo(eidos.dunamis,arche.prop.title,eidos.fullName)
			const tropos = p =>p.aliases = [
				"className",
				"directName",
				"inverseName",
			].map(t=>eidos[t]).filter(Boolean);
			await app.fileManager.processFrontMatter(eidos.dunamis,tropos);
		}
		{
			const kytos = "# あらまし"
			const hyle = [
				"examples",
				"inFirstOrderLogic",
				"quantification",
			].map(d=>d in eidos? `### ${d}\n${eidos[d]}`: null)
			
			eidos.scopeNote ??= "!" + await Promise
				.resolve(noema.crm_objects[eidos.domain])
				.then(getLinkText);
			hyle.unshift(kytos,eidos.scopeNotes);
			const tropos = t=>t.replace(kytos,hyle.filter(Boolean).join("\n"));
			await app.vault.process(eidos.dunamis,tropos);
		}
	};
	
	
};
if(typeof isModule === "undefined") noesis();
