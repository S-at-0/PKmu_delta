const noesis = async function(){
	await sleep(1000)
	const [
		{arche, transDunamis, Melete: {assignTo}},
		noesis,
		kytosTesserae,
		exoTally,
		...kinesis
	] = await syzygy('26-251-1989');
	const eidos = await Promise
		.resolve(transDunamis()("cidoc_crm_v7.1.3.json"))
		.then(d=>app.vault.read(d))
		.then(JSON.parse);
	const tessera = kytosTesserae();
	Object.assign(eidos.crm_objects, eidos.crm_PoP)
	for(const [khora,noema] of Object.entries(eidos.crm_objects)){
		noema.tipos = 
			"className" in noema? "class":
			"propertyOfProperty" in noema? "typedProp":
			khora in eidos.crm_PoP? "PoP":"property";
		const morphe = {
			eidos: tessera[noema.tipos],
			eidolon: "",
		}; noema.dunamis = await exoTally(morphe);
		
		await assignTo(noema.dunamis,arche.prop.title,noema.fullName)
		const tropos = p =>p.aliases = [
			"className",
			"directName",
			"inverseName",
		].map(t=>noema[t]).filter(Boolean);
		await app.fileManager.processFrontMatter(noema.dunamis,tropos);
	}
	
	const topics = [
		"writing down overlooks",
		"linkage constructuring"
	]
	for (const i in topics){
		const notice = new Notice(topics[i],0);
		let counter = 0;
		for(const [khora, noema] of Object.entries(eidos.crm_objects)){
			notice.setMessage(topics[i]+ ": " + khora +` (${++counter})`);
			try{await kinesis[i].call({eidos: eidos.crm_objects},khora,noema)}
			catch(e){new Notice(khora+e.message,0)};
		}
		notice.setMessage(topics[i] + ": done");
		notice.messageEl.style.color = "green";
	}
};
if(typeof isModule === "undefined") noesis();
