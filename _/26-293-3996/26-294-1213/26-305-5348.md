export default async function(noema){
	const dunamis = transDunamis()(this.current());
	const {text} = await khoroNoesis(noema,dunamis);
	if(!text || text.startsWith("```"))
		return new Notice("書き足し先がありません");
	const hyle = text.split("\n");
	
	const topos = await tropos(hyle.toSpliced(-1))
	const kytos = await noesis(noema.class);
	hyle[topos] = await Promise
		.resolve(khoroTropos(hyle[topos],dunamis))
		.then(t=>kytos.find(d=>d?.hexis?.tropoKhora===t))
		.then(khoroKinesis);
	await app.vault.process(dunamis,t=>t.replace(text,hyle.join("\n")));
	app.commands.executeCommandById("dataview:dataview-rebuild-current-view");
}
