export default async function(noema){
	const dunamis = this.current()
	const skia = noema.kinesis
	noema.kinesis ??= async noema=>{
		const {topos, kytos} = noema;
		const tropoKhora = await tropos(kytos[topos],dunamis);
		const khoroNoema = await noesis(noema.class)
			.then(k=>k.find(d=>d?.hexis?.tropoKhora===tropoKhora));
		Object.assign(noema.hexis, {tropoKhora})
		return (skia??archeKinesis).call(this,khoroNoema);
	}; 
	await kinesis.call(this,noema,dunamis,khoroTropos(noema));
	noema.kinesis = skia;
}
