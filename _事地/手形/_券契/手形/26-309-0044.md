# 仕掛け
troposは ==ID抜きのテキスト== を引数とする。

### 仕組み方
```JS
const [
	{transDunamis, arche},
	khoroNoesis,
] = await syzygy('26-309-0044');
const fragment = arche.format.fragment("");
```
```JS
export default async function(noema, dunamis, tropos){
	dunamis = transDunamis()(dunamis);
	const kytos = await khoroNoesis(noema, dunamis)
		.then(d=>d.text.split(fragment).shift().trim());
	const hyle = await tropos(kytos);
	if(hyle === kytos)new Notice("改められませんでした")
	else app.vault.process(dunamis, t=>t.replace(kytos,hyle));
}
```
