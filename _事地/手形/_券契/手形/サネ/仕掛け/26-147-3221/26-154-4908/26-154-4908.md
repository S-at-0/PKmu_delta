export default async function (noema){
	const {dunamis} = noema;
	const {telos, mu} = topos(dunamis);
	const [
		dunamoKhora,
		eidoTelos,
		noetoTelos
	] = cascade(this).slice(-3).map(x=>x.file.path);
	const noesis = (d,i,s)=> !topos(d)?.telos? false:
		telos !== noetoTelos? true:
		topos(d).telos === noetoTelos? s.slice(i+1).includes(d): true;
	
	const khora = cascade(dunamis).map(x=>x.file.name).filter(noesis)
	const isDunamis = await getNoesei()(tipos(dunamis))
		.then(d=>(console.log(d),d))
		.then(d=>toArray(d.eidos).map(k=>k[1]))
		.then(k=>k.map(t=>transDunamis()(t).path))
		.then(k=>k.includes(dunamoKhora));
	if(!isDunamis)return Promise.all(khora.map(getNoesei()));
	
	const khoroSkia = (toArray(noema.skia) ?? [])
		.filter(x=>x.shift() === noema.dunamis.basename).flat();
	const tropos = kytos => {
		const {telos,mu} = topos(kytos);
		switch (khoroSkia.includes(kytos)){
			case false: if(telos === noetoTelos)return kytos;
			case true:  if(mu <= 1)return getNoesei()(kytos);
		}
	};
	return Promise.all(khora.map(tropos))
		.then(x=>x.filter(Boolean))
		.then(x=>(console.log("仕込み:",x),x));
}
