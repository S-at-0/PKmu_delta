export default async function (noema){
	const {taxis={},hexis} = Object(noema?.morphe);
	if(!Object.keys(taxis).length){
		new Notice("taxis is inValid.",0);
		Object.assign(taxis,{title:"",name:"",fields:[]});
	}
	const skia = recentView().leaf;
	const khora = app.workspace.getLeaf("tab");
	await khora.setViewState({type,active:true});
	const {containerEl, formEditor} = khora.view;
	const eidos ={
		target: containerEl,
		props:{
			definition:taxis, app,
			onPreview: hyle=> mf.openForm(taxis,{values:hexis})
				.then(d=>(new Notice(JSON.stringify(d, null, 2)),d))
				.then(d=>d.status==='ok'? d.data:null)
				.then(d=>{if(d)Object.assign(hexis,d)})
				.catch(console.error),
			onSubmit: hyle=>{
				Object.assign(noema.morphe.taxis,hyle);
				ousioKinesis(noema)
					.then(_=>new Notice("フォームは仕立てられました"));
				khora.detach();
			},
			onChange:_=>app.workspace.requestSaveLayout(),
			onCancel:_=>{
				khora.detach();
				app.workspace.revealLeaf(skia);
			},
		}
	};
	formEditor.$destroy();
	khora.view.formEditor = new formEditor.constructor(eidos);
}
