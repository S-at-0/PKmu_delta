	for(const {dunamis, ...hyle} of khora){
		const {mu} = Object(topos(dunamis));
		if(mu <= 1)siftedAssign(morphe, hyle.morphe)
		else siftedAssign(morphe.hexis, hyle);
	}
	const eidolon = new Set();
	const {fields} = morphe.taxis;
	const kytos = [];
	while(fields.length)kytos.push(fields.pop());
	for(const hyle of kytos){
		if(!eidolon.has(hyle.name))fields.unshift(hyle);
		eidolon.add(hyle.name);
	}
