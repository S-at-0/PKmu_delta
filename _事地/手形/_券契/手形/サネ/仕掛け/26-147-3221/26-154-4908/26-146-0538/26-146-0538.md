export default async function (noema){
	const {dunamis} = noema;
	const {mu} = Object(topos(dunamis));
	const {hexis,taxis} = await getMorphen(noema);
	const {file, ...khora} = transDunamis(true)(dunamis);
	for(const kytos of Object.keys(file.frontmatter))
		delete khora[kytos];
	siftedAssign(hexis,khora);
	
	if (mu <= 1){
		await troposTessaron(noema);
		noema.morphe = {hexis,taxis};
		if(await qa.yesNoPrompt(
			"フォームを仕立てますか?",
			"その場合、仕掛けのみ取り込まれます。"
		))return await troposMorphei(noema);
	}
	const telos = mu<=1? noema.morphe.hexis:noema;
	await mf.openForm(taxis,{values:hexis})
		.then(d=>d.status==='ok'?d.data:null)
		.then(d=>{if(d)siftedAssign(telos,d)});
	return noema;
};
