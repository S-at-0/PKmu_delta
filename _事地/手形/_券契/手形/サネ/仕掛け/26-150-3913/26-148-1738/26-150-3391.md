export default async function (noema){
	const {telos, mu} = noema.topos;
	noema = Object.getPrototypeOf(noema);
	const antiTelos = {
		"eidos": "skia",
		"skia": "eidos",
	}[telos];
	const isDunamis = function(x){
		return x.every(topos) && 
		cascade(x[0])
			.map(p=>p.file.name)
			.includes(x[1]);
	}(noema[telos][mu]);
	
	const kinesis = {
		shiftUp: {
			textContent:"↑",
			noesis: x=>{if(mu)[x[mu-1],x[mu]]=[x[mu],x[mu-1]]},
		},
		shiftDown:{
			textContent:"↓",
			noesis: x=>{if(noema[telos][mu+1])[x[mu+1],x[mu]]=[x[mu],x[mu+1]]},
		},
		siftUp:{
			textContent:"⌘",
			noesis: x=>(noema[antiTelos]??=[]).push(...x.splice(mu,1)),
		},
		siftDown:{
			textContent:"×",
			style:{color:"red"},
			noesis: x=>x.splice(mu,1),
		},
	};
	
	for(const [khora,kytos] of Object.entries(kinesis)){
		const {noesis,...hyle} = kytos
		const onclick = async _=>{
			const isConfirmed = !isDunamis 
				|| khora !== "siftDown"
				|| await qa.yesNoPrompt("誠に取り除きますか？")
					.catch(console.error);
			if(!isConfirmed)return null;
			await noesis(noema[telos]);
			await ousioKinesis(noema);
		}
		kinesis[khora] = Object.assign(
			createEl("button"),
			{...hyle,onclick});
	}
	const {siftUp,siftDown,shiftUp,shiftDown} = kinesis;
	const kytos = [isDunamis? siftUp:siftDown]
	if(telos === 'eidos')kytos.push(shiftUp,shiftDown);
	const div = createDiv();
	kytos.forEach(b=>div.appendChild(b))
	return div;
}
