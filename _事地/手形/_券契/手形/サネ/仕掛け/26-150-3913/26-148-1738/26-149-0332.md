export default async function(noema){
	if(!noema[noema.topos.telos])return null;
	const eidos = Array.from(noema[noema.topos.telos]);
	const noesis = hyle => !dv.page(hyle)? hyle: dv.fileLink(
		dv.page(hyle).file.path,
		false,
		dv.page(hyle)[arche.prop.title]
	);
	
	for(const [mu,kytos] of eidos.entries()){
		const [tessera, tally] = [...kytos].map(noesis);
		Object.assign(noema.topos,{mu});
		eidos[mu] = [
			tessera, 
			arche.format.est,
			tally, 
			await kinesis(noema)
		];
	};
	return [noema.topos.telos, eidos];
}
