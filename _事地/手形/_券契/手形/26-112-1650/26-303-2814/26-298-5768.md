export default async function(dunamis,isRetrorse){
	const eidoTelos = cascade(this).pop().file.path;
	const eidoTropos = d=> topos(d)?.telos === eidoTelos;
	
	const kinesis = hyle => {
		hyle = transDunamis(true)(hyle);
		const khora = isRetrorse
			? [...new Set(cascade(hyle))]
			: retrorse(cascade)(hyle);
		const tropos =async noema=>{ if(!topos(noema)?.mu<=1)return null;
			const eidos = [hyle,noema]; if(isRetrorse)eidos.reverse();
			const eidoKhora = eidos.map(eidoTropos).join();
			switch(eidoKhora){
				case "true,true" : return noema;
				case "true,false": 
					const khora = cascade(eidos[1]).map(d=>d.file.name)
					const dimer = Array(khoroTaxis(khora)[1]);
					return dimer.includes(eidos[0].file.name)? noema:null;
				default:
					const kytos = await getNoesei()(eidos[1])
					.then(d=>d?.skia || [])
					.then(k=>toArray(k).map(t=>t[1]));
				if(kytos.includes(eidos[0].file.name))return noema;
			}
		}
		return Promise.all(khora.map(tropos)).then(k=>k.filter(Boolean));
	}
	return await relayTree(dunamis,kinesis);
}
