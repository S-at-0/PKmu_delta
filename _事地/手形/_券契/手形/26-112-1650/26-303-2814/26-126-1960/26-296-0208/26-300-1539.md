export default async function(noema,khoroKinesis=kinesis){
	const khoroNoesis = noema.isNoesis
		? await syzygy(noema.dunamis)
		: khoroNoesis;
	const khora = await khoroKinesis(noema)
		.then(khoroNoesis.bind(this))
		.then(x=>[x].flat());
	const tropoNoema = await Promise
		.all(khora.map(d=>transDunamis()(d)?getLinkText(d):d))
		.then(k=>k.join());
	noema = Object.create(noema);
	noema.hexis = {...noema.hexis, delta:1, tropoNoema};
	return tropos(noema);
}
