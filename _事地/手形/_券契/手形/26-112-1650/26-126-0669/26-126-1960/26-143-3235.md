export default async function (noema){
	const {tropoKhora,eidolon} = noema.hexis;
	const dunamis = transDunamis()(this.current());
	const ousia = messengers(tipos(dunamis)).shift();
	const khora = arche.format.fragment(eidolon[ousia.file.name]);
	const kytos = await getSections(dunamis)
		.then(x=>x?.blocks?.[khora.replace('^','')]?.text)
	if(!kytos || kytos.startsWith("```"))
		return new Notice("書き足し先がありません");
	const hyle = kytos.split("\n");
	const skia = hyle.toSpliced(-1).map((t,i)=>String(i+1)+"個目: "+t);
	const topos = await qa.suggester(skia,skia.map((d,i)=>i));
	const tropoNoema = await tropoNoesis(noema)
		.then(tropos)
		.then(getLinkText);
	hyle[topos] = `[${tropoKhora}:: ${tropoNoema}]`;
	await app.vault.process(dunamis,t=>t.replace(kytos,hyle.join("\n")));
	app.commands.executeCommandById("dataview:dataview-rebuild-current-view");
}
