# あらまし
- 二度記されたものはその事地のみを探る
- 型ならば:
	1. 型のみカタを探る
	2. それぞれを型とする事地を返す
- 型でなければ:
	- その型が元のそれと等しければ、再帰探索
	- 異なる、またはisNoesis=trueならば返す時モジュールとして扱う

export default async function kinesis(noema,hyle){
	hyle = transDunamis()(hyle || noema.dunamis);
	const khoroTipos = tipos(noema.dunamis).file.path
	
	const noetoTropos = (d,i,s) =>
		s.slice(0,i).includes(d)? null:
		s.slice(i+1).includes(d)? transDunamis()(d): d;
	const tipoTropos = hyle => x => topos(x)?.mu>1? null: getNoesei(false)(x)
		.then(d=>toArray(new Set(d?.skia)).map(k=>k?.[1]))
		.then(k=>k.includes(hyle)? x.file.name: false);
	const kinesis = async hyle => 
		typeof hyle !== "string"? []:
		tipos(hyle)?.file?.path===khoroTipos? 
			cascade(hyle,false).map(x=>x.path).map(noetoTropos):
		topos(hyle)?.mu<=1? await Promise
			.all(retrorse(cascade)(hyle).map(tipoTropos))
			.then(k=>k.filter(Boolean)):
		[];
	const noesis = async (hyle,kytos) =>{
		const isNoesis = tipos(hyle)?.file?.path!==khoroTipos
			|| await getNoesei(true)(hyle).then(d=>d.hexis.isNoesis);
		const dunamis = 
			typeof hyle!=="string"? [hyle]:
			topos(hyle)?.mu<=1? retrorse(tipos)(hyle).filter(d=>!isMessenger(d)):
			isNoesis? await syzygy(transDunamis()(hyle))
				.then(f=>f.call(this,noema)): [];
		return Promise.all(kytos)
			.then(k=>[...k.flat(),...dunamis.map(transDunamis())]);
	}
	const khora = [...await relayTree(hyle.basename,kinesis,noesis)];
	
	const {
		tropoKhora,
		rho=0, kappa=0,
	} = noema.hexis
	const skia = dv.pages("")
		.where(p=>p[tropoKhora])[tropoKhora]
		.path.array();
	const tropos = d => d && skia.filter(t=>t===d).length < rho + kappa
	return !noema.hexis.kappa? khora: 
		khora.map(d=>transDunamis()(d)?.path).filter(tropos);
}
