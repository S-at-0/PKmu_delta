export default async function(noema){
	const {eidolon,tropoKhora} = Object(noema.hexis);
	const ousia = transDunamis()(this.current());
	const dunamis = messengers(tipos(ousia)).shift();
	const khora = arche.format.fragment(eidolon[dunamis.file.name]);
	const kytos = await getSections(ousia)
		.then(x=>x.blocks[khora.replace('^','')].text)
		.then(t=>t.replace(khora,"").trim())
		.catch(console.error);
	if(!kytos) return new Notice("書き足し先がありません");
	
	const hyle = [
		kytos,
		`[${tropoKhora}:: ]`,
	].join("\n");
	await app.vault.process(ousia,t => t.replace(kytos,hyle));
}
