export default async function(noema,dunamis){
	dunamis = transDunamis(true)(dunamis);
	const {eidolon} = noema.hexis
	
	const kytos = messengers(tipos(dunamis))
		.find(p=>dv.date(p[time]) < dv.date(dunamis[time]))

}
