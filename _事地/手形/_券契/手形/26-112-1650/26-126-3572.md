export default async function (noema){
	const {dunamis, hexis, substrate} = noema;
	const {isSyzygy} = hexis;
	const khora = await Promise
		.all(cascade(dunamis).map(getNoesei(true)))
		.then(s=>s.filter(Boolean));
	if(isSyzygy)khora.forEach(x=>x.hexis.isMneme = false);
	return noesis({khora, ...substrate});
}
