# 形
```JSON
"PROP/name":{
	"key": "PROP/name",
	"mode": 2,
	"files":[
		"path1",
		"path2"
	]
}
```
# 組み方
`キー/呼び名` で構成。
1. 呼び名が **ファイル** 名
	- CONFIG
	- SNIPPET
2. 呼び名が ==フォルダ== 名
	- THEME
	- PLUGIN_DATA
	- PLUGIN_MAIN
また、`files`にはいずれも`.obsidian/`以降のパスが記される。
### プラグイン名前の違い
下の名前の事地からあるものだけがが含まれる。
- DATA: `data.json`
- MAIN: `main.js` `manifest.json` `styles.css`
`tracked_files.json`などの他ファイルは無視される。これが従来のhidden filesに1つだけかかった所以か
***
# あらまし
Self-hosted LivesyncのCustom Synchronisationは、 デフォルトは全てSelectiveになる。しかし、限られた事柄のみを紐付けるなら、全てIgnoredから始めるのが妥当になる。
今の所公式はそのような振る舞いを備えないため、`data.json`を直に書き換え全て切る。

>[!warning] 完全な動作を保証しません
> この仕組みは見た目による憶測で組み立てました。
> また、元の事柄を破壊的に上書きします。
> 
> あくまでも人の目を通すよう忘れずにお願いします。
# 仕掛け
### 仕組み方
```js
const kinesis = async function(){
	//await syzygy('26-214-0905');
	const {configDir, adapter} = app.vault;
	const khora = configDir + "/plugins/obsidian-livesync/data.json";
	const noema = await app.vault.adapter.read(khora)
		.then(JSON.parse)
		.catch(console.error);
	if(!noema)return new Notice("事柄が見つかりませんでした");
	else noema.pluginSyncExtendedSetting??={};
	
	const tropos = eidos => hyle => {
		if(![hyle].flat().filter(Boolean).length)return {};
		const kytos = (Array.isArray(hyle)? hyle[0]: hyle)
			.split("/").reverse()[Array.isArray(hyle)? 1:0];
		const key = [eidos,kytos].join("/");
		const files = [hyle].flat().map(t=>t.replace(configDir+"/"),"");
		return {[key]:{key,files,mode:2}};
	}
	const kytoNoesis = topos=> app.vault.adapter.list(topos)
		.then(x=>Promise.all(x.folders.map(x=>app.vault.adapter.list(x))))
		.then(d=>d.map(x=>x.files));
	const ousia = {
		"CONFIG": (await app.vault.adapter.list(configDir)).files,
		"SNIPPET":(await app.vault.adapter.list(configDir+"/snippets")).files,
		"THEME": await kytoNoesis(configDir+"/themes"),
		"PLUGIN_DATA": await kytoNoesis(configDir+"/plugins")
			.then(x=>x.map(s=>s.filter(p=>p.endsWith("data.json")))),
		"PLUGIN_MAIN": await kytoNoesis(configDir+"/plugins")
			.then(x=>x.map(x=>x.filter(
				p=>[
					"main.js","manifest.json","styles.css"
				].some(t=>p.endsWith(t))
			))),
	};
	for (const [eidos,hyle] of Object.entries(ousia)){
		new Notice(eidos);
		Object.assign(
			noema.pluginSyncExtendedSetting,
			...hyle.map(tropos(eidos))
		);
	}
	await app.vault.adapter.write(khora, JSON.stringify(noema, null, 2));
	await app.plugins.plugins.quickadd.api.utility.setClipboard(JSON.stringify(ousia, null, 2));
	
	new Notice("Settings overwritten!");
};
app.plugins.disablePlugin("obsidian-livesync").then(kinesis)
```
