const noesis = async function(){
	const {
		transDunamis,
		Melete: {getSections, assignTo},
	} = await self.require.import("SrcBarn/DatumOvule.js");
	const tropos = async function(dunamis){
		let hyle = await app.vault.read(dunamis);
		const tropoTipos = t => [
			'```dataviewjs',
			`dv.view("KhoroScope","${t}")`,
			'```',
		].join("\n");
		const eidos = await getSections(dunamis);
		const isEide = !hyle
			.split("***")[1]
			.split(/^# 仕訳所\s*$/m)
			.shift()
			.match(/^\^gzb/m);
		if(!isEide){
			const kytos = eidos.blocks[
				Object.keys(eidos.blocks).find(x=>x.includes("gzb"))
			].text;
			const noema = tropoTipos("25-292-5767");
			hyle = hyle.replace("^gzb0","^gzb1").replace(kytos,noema);
		}
		if(hyle.toLowerCase().includes("khoroscope"))return hyle
			.replaceAll("khoroScope","KhoroScope")
			.replaceAll("SrcBarn/KhoroScope","KhoroScope");
		const kytos = eidos.headings
			.find(d=>d.heading.includes("仕訳所"))
			.text;
		const noema = [
			kytos.split('\n```').shift(),
			tropoTipos("26-178-4761"),
			isEide && "### 見出し分け",
			isEide && tropoTipos("26-191-0573")
		].filter(Boolean).join("\n");
		hyle = hyle.replace(kytos,noema);
		
		return hyle
			.split(/^#{2,3} 全ての事柄\s*$/m)
			.toSpliced(1,Infinity,tropoTipos("26-178-5622"))
			.join("### 全ての事柄\n");
	};
	
	const khora = dv.pages('"_"').array();
	let count=0;
	for(const dunamis of khora.map(transDunamis())){
		try{
			const hyle = await tropos(dunamis);
			if(await app.vault.read(dunamis) === hyle)continue;
			await app.vault.modify(dunamis,hyle);
			count++
			if(transDunamis(true)(dunamis).tags.every(x=>!x.includes("inDex")))
				await assignTo(dunamis, "tags", "inDex");
			console.log("processed: " + dunamis.name);
		}
		catch(e){
			console.error(e)
			console.log(dunamis.name + " looks not tobe a turret. skipping");
		}
	} new Notice("all processes have done! "+count+" files changed");
};
if(typeof isModule === "undefined") noesis();
