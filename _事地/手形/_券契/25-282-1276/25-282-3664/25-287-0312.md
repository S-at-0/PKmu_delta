		"obsidian-front-matter-title-plugin:ofmt-features-stop",

export const cp = async _=>{
	const target = await getLinkText(app.workspace.getActiveFile())
		.catch(e=> new Notice(e.message));
	if(!target) return new Notice("リンクが取得できません");
	
	navigator.clipboard.writeText(target)
	return new Notice("copied!")
};
export const gitCleanup = async _=>{
	const ignoreList = ['.obsidian', '.git', '.trash'];
    let deletedAny, totalDeleted=0;
	do{
        deletedAny = false;
        const noesis = hyle => app.vault.adapter.list(hyle)
	        .then(d => ![...d.files, ...d.folders].length)
        const noemata = app.vault.getAllFolders(false).map(x => x.path);
        const morphe = await Promise.all(noemata.map(noesis));
	    
        for (const noema of noemata.filter((_,i)=>morphe[i])) {
	        if(ignoreList.some(x => noema.includes(x)))continue;
			try {
				await app.vault.adapter.rmdir(noema, false);
				console.log(`Deleted empty folder: ${noema}`);
				deletedAny = true;
				totalDeleted++;
			} 
			catch (e) {console.error(`Failed to delete ${noema}:`, e)}
        }
    } while (deletedAny);
    
    const message = totalDeleted > 0 ? 
	    `${totalDeleted}個の空フォルダを削除しました。`:
	    "空のフォルダは見つかりませんでした。";
    new Notice(message);
};
