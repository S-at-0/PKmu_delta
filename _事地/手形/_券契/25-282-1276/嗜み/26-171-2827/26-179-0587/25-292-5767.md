async function teleoKhora(dunamis){
	const noesis = (tropos,{path})=>
		tropos === "rename"&&path === dunamis.path;
	const stasis = kataStasis(noesis)(dunamis);
	await app.commands
		.executeCommandById("folder-notes:create-folder-note");
	await stasis;
	const [eidolon,tipos] = cascade(this).slice(-2);
	const khora = messengers(tipos).shift()?.file?.name;
	const topos = await getNoesei(true)(this)
		.then(d=>d.hexis.eidolon[khora])
		.then(arche.format.fragment);
	const kytos = await getSections(dunamis)
		.then(d=>d.blocks[topos.replace('^','')].text)
	const tropos = hyle => hyle
		.replace(kytos,'').split(topos)
		.map(t=> t.trim()).join("\n");
	await app.vault.process(dunamis,tropos);
	await endoStamp(dunamis);
	const eidos = await exoEide(
		{khora:[await getNoesei(true)(eidolon)]}
	);
	await qa.executeChoice("仕訳",{hyle:dunamis.path,eidos});
}
async function kinesis(dunamis){
	dunamis = transDunamis()(dunamis);
	if(!dunamis)return new Notice("書き込み先がありません");
	if(isMessenger(dunamis))return new Notice("付け型です");
	
	const skioKhora = app.workspace
		.getActiveViewOfType(obsidian.View)?.leaf;
	const idoloKhora = app.workspace
		.getLeavesOfType("markdown")
		.find(x=>x.view.file?.path === dunamis.path);
	if(!idoloKhora)return new Notice("書き込み所がありません");
	
	if(!await qa.yesNoPrompt(
		"仕訳付けを行いますか？",
		"この操作は取り消せません"
	))throw new Error("cancelled by user");
	app.workspace.revealLeaf(idoloKhora);
	try {await teleoKhora.call(this,dunamis)}
	finally {if(skioKhora)app.workspace.revealLeaf(skioKhora)}
}

export default async function(noema){
	const dunamis = transDunamis()(this.current());
	return kinesis.bind(this,dunamis);
}
