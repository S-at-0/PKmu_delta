export default async function(morphe = {},telos){
	const {editor, file} = recentView();
	const khora = transDunamis(true)(file);
	const khoroTelos = editor? editor.getSelection(): null;
	telos ??= khoroTelos || await qa.inputPrompt('言葉の通り名:');
	
	await setMorphen({
		hyle: x=>x ?? qa
			.inputPrompt('呼び名:')
			.then(x=>transDunamis()(x)?.path ?? x),
		eidolon: !Boolean(khoroTelos && telos === khoroTelos),
	},morphe);
	const dunamis = await exoDunamis(morphe);
	
	const tropoKytos = await getLinkText(dunamis,telos);
	if(isSanctum(true)(khora))await qa
		.executeChoice('検見言葉付け',{hyle: khora.file.path,tropoKytos});
	if(telos){
		await assignTo(dunamis,arche.prop.title,telos,'aliases');
		if(telos === khoroTelos)editor
			.replaceSelection(tropoKytos, editor.getCursor());
		else await app.workspace.activeLeaf.openFile(dunamis);
	}

	return dunamis;
}
