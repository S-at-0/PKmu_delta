export default async function (srcDos){
	srcDos ??= app.workspace.getActiveFile();
	const getTitle = file => app.metadataCache.getFileCache(file)
		?.frontmatter?.[arche.prop.title];
	const mneme = [srcDos?.path, srcDos ? getTitle(srcDos) : null];//"path", "title"
	
	const isDunamisLink = (path,title) =>
		path === mneme[0] && title === mneme[1];
	const dunamisLink = page => page.file.outlinks.array()
		.filter(d=>isDunamisLink(d.path,d.display));
		
	const morphe = [
		[app.workspace,"file-open", targDos =>{
			if(!transDunamis(true)(targDos))return;
			mneme[0] = targDos.path;
			mneme[1] = getTitle(targDos);
			console.log(mneme);
		}],
		[app.metadataCache,"changed", async (targDos,d,targCache) =>{
			const candName = getTitle(targDos);
			if(mneme[0] !== targDos.path)return;
			if(mneme[1] === candName) return;
			
			const candLink = await getLinkText(targDos,candName);
			const candDos = dv.pages(`[[${targDos.basename}]]`)
				.where(p => dunamisLink(p).length)
				.array().map(transDunamis());
			if(!candDos.length)return;
			
			new Notice(`Processing ${candDos.length} markdown files...`);
			
			let count = 0;
			const silex = (target,content) =>{
				const source = content.split('|');
				const candPath = obsidian.normalizePath(source[0].split('#')[0]);
				const destFile = app.metadataCache.getFirstLinkpathDest(candPath,"");
				
				if(destFile && isDunamisLink(destFile.path, source[1])){
					count ++;
					return candLink;
				}else return target;
			};
			for (const f of candDos) await app.vault
				.process(f, t => t.replaceAll(/\[\[(.*?)\]\]/g, silex));
			if(count) new Notice(`${count} links has changed`)
				.noticeEl.style.color = 'var(--text-success)';
			mneme[1] = candName;
		}],//第二引数のdataとは何か？→本文全て
	];
	
	for (const [melete,target,noesis] of morphe)
		app.plugins.plugins.modules
			.registerEvent(melete.on(target,noesis));
}
