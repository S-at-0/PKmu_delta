```JS
const polymerase = (a,d,i,s) => {
	const {residue, terminum} = d.substrate;
	const {isMneme, isMultiLine, isOverRidable,} = Object(d?.hexis);
	const {isSpaceNeeded} = s
		.slice(0,i)
		.findLast(x=> x.substrate.residue)
		?.hexis || {};
	const fragment = arche.format.fragment(terminum);
	return a + "\n"
		+ (isSpaceNeeded&&!isOverRidable?"\n":"") + residue
		+ (isMneme? (isMultiLine? "\n":" ")+fragment : "");
}

export default async function ({khora, ...substrate}){
	substrate.terminum ??= 1;
	const noetoTopos = cascade(this,false).pop().path;
	
	console.log("polymerase called:",khora);
	for(const noema of khora){
		const tessera = transDunamis()(tipos(noema.dunamis));
		if (topos(tessera)?.telos !== noetoTopos){
			console.warn("looks not to be noetoTessera:", noema);
			continue;
		};
		substrate.residue = await syzygy(tessera)
			.then(f => f({...noema,substrate}))
			.catch(console.error);
		noema.substrate = {...substrate};
		console.log(tessera.basename,noema.substrate);
		if (noema.hexis.isMneme && substrate.residue) substrate.terminum++;
	}
	return khora.reduce(polymerase,"").trim();
}
```
