export default function (event,view){
	const khoroNoema = event.target.closest('a.internal-link');
    if (khoroNoema)return {
		label:(khoroNoema.getAttribute('data-href')
			|| khoroNoema.getAttribute('href')||"").trim(),
		alias:(khoroNoema.innerText || "").trim(),
    }
	const isKhorai = [
		'.cm-underline',
		'.cm-hmd-internal-link',
		'.cm-link',
	].some(x => event.target.closest(x))
		&& Boolean(view)
		&& Boolean(app.workspace.activeEditor);
	if(!isKhorai)return null;
	
	const khora = app.workspace.activeEditor.editor;
	const {clientX:x,clientY:y} = event.changedTouches?.[0]??event;
	if([x,y].some(isNaN))return null;
	
	const hyloTopos = view.posAtCoords({x,y});
	const khoroTopos = khora.offsetToPos(hyloTopos);
	const {
		text: label, displayText: alias,
	} = khora.getClickableTokenAt(khoroTopos);
	return {label,alias};
}
