const noesis = {
	newTab: event => {
		const khora = event?.target?.closest('.workspace-tabs');

		const kinesis = leaf => {
			const isDunamis = !isSkia

				&&!leaf.containerEl.classList.contains('mod-hidden');
			if(isDunamis){
				khora.ownerDocument.defaultView?.focus();
				app.workspace.setActiveLeaf(leaf, { focus: true });
				isSkia = true;
			}
		};

	},

const getButton = ([isMobile,target]) => isMobile?
	app.mobileNavbar?.containerEl?.querySelectorAll(target):
	document.querySelectorAll(target);

const kinesis =_=>{
	const [plusButton,searchButton,headerPlus] = [
		[true,'.mobile-navbar-action-new-tab'],
		[true,'.mobile-navbar-action-quick-switcher'],
		[false,'.workspace-tab-header-new-tab .clickable-icon'],
	].map(getButton);
	
	const ousia = [
		[plusButton, 'click', teleoDunamis],
		[headerPlus,'click',noesis.combi],
		[plusButton,'contextmenu', endoCascade],
		[searchButton, 'contextmenu', exoTurret],
		[headerPlus,'contextmenu', noesis.newTab],
	]
	
	for (const target of ousia){
		const [candEl,targType,srcEvent] = target;
		if(!candEl)continue;
		
		const targEvent = event => {  
			event.stopImmediatePropagation();  
			event.preventDefault();  
			srcEvent(event); 
		}; 
		for(const targEl of Array.from(candEl))
		targEl?.addEventListener(targType, targEvent, true);
	}
}
export default _=>{
	kinesis();
	app.workspace.on('layout-change',kinesis);
	console.info('mounted buttons.');
}
