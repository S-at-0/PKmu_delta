export default async function(khora,noema){
	const {
		dunamis, tipos:khoroTipos,
		subClassOf, subPropertyOf,
		domain, range, directName: tropoKhora,
	} = noema;
	const kinesis = (dunamis,...khora) => Promise
		.resolve(khora.flat(Infinity).filter(Boolean))
		.then(k=>k.map(t=>getLinkText(this.eidos[t].dunamis)))
		.then(k=>Promise.all(k))
		.then(k=>assignTo(dunamis,arche.prop.cascade,k));	
	if(khoroTipos !== "PoP"){
		await kinesis(dunamis,...[subClassOf,range].flat().filter(Boolean));
		if(khoroTipos === "class"||!domain?.length)return true;
		for(const k of domain)await kinesis(this.eidos[k].dunamis,khora);
	}
	
	const {hexis} = await getNoesei(true)(tipos(dunamis));
	const [[delta,nu],[rho,kappa]] = noema.quantification
		.match(/\((.*)\)/).shift().replaceAll("n","0")
		.replaceAll("(","").replaceAll(")","")
		.split(":").map(t=>t.split(",").map(Number))
		.map(([a,b])=>[a, b && b-a+1]);
	const morphe = {
		...hexis,
		dunamis,tropoKhora,
		delta,nu,rho,kappa,
		necessary: Boolean(delta),
	}
	await ousioKinesis(morphe);
}
